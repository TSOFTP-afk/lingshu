"""
共享依赖注入 — 提供 BookstoreStore 实例和认证辅助函数。

状态存储在 FastAPI app.state 上，通过 Depends 提供依赖注入，
避免模块级全局变量，利于测试和多实例场景。
"""
from __future__ import annotations

import logging
import secrets
from typing import Any

from fastapi import Header, HTTPException, Request

from lingshu_api.services.book_sync import BookSyncService
from lingshu_api.services.db_bridge import WorkspaceDB
from novel_app.bookstore import BookstoreStore, CloudBookStore, _bearer_token, _public_user

# ── setter：由 main.py 在 create_app 时调用，写入 app.state ──────────

def set_store(app, store: BookstoreStore) -> None:
    """设置 store 实例到 app.state（在 create_app 时调用）。"""
    app.state.store = store


def set_cloud_store(app, cloud_store: CloudBookStore) -> None:
    """设置 CloudBookStore 实例到 app.state（在 create_app 时调用）。"""
    app.state.cloud_store = cloud_store


def set_sync_service(app, service: BookSyncService) -> None:
    """设置 BookSyncService 实例到 app.state（在 create_app 时调用）。"""
    app.state.sync_service = service


# ── FastAPI 依赖：通过 Request.app.state 读取 ──────────────────────

def get_store(request: Request) -> BookstoreStore:
    """FastAPI 依赖：获取 BookstoreStore 实例。"""
    store = getattr(request.app.state, 'store', None)
    if store is None:
        raise HTTPException(status_code=500, detail="Store 未初始化")
    return store


def get_cloud_store(request: Request) -> CloudBookStore:
    """FastAPI 依赖：获取 CloudBookStore 实例。"""
    cloud_store = getattr(request.app.state, 'cloud_store', None)
    if cloud_store is None:
        raise HTTPException(status_code=500, detail="CloudStore 未初始化")
    return cloud_store


def get_sync_service(request: Request) -> BookSyncService:
    """FastAPI 依赖：获取 BookSyncService 实例。"""
    sync_service = getattr(request.app.state, 'sync_service', None)
    if sync_service is None:
        raise HTTPException(status_code=500, detail="SyncService 未初始化")
    return sync_service


def get_optional_user(request: Request, authorization: str = Header(default="")) -> dict[str, Any] | None:
    """解析 Authorization 头，返回用户信息或 None。"""
    token = _bearer_token(authorization)
    if not token:
        return None
    return _public_user(get_store(request).get_user_by_token(token))


def require_login(request: Request, authorization: str = Header(default="")) -> dict[str, Any]:
    """要求已登录，返回用户信息。"""
    token = _bearer_token(authorization)
    if not token:
        raise HTTPException(status_code=401, detail="未登录")
    user = get_store(request).get_user_by_token(token)
    if not user:
        raise HTTPException(status_code=401, detail="登录已过期")
    return user


def require_ai_access(request: Request, authorization: str = Header(default="")) -> dict[str, Any]:
    """AI 功能统一门禁：所有会调用模型或配置模型的入口必须先登录。"""
    token = _bearer_token(authorization)
    if not token:
        raise HTTPException(status_code=401, detail="请先登录后使用 AI 功能")
    user = get_store(request).get_user_by_token(token)
    if not user:
        raise HTTPException(status_code=401, detail="登录已过期，请重新登录后使用 AI 功能")
    return user


def require_author(request: Request, authorization: str = Header(default="")) -> dict[str, Any]:
    """要求已登录且为作者身份。"""
    user = require_login(request, authorization)
    if not int(user.get("is_author") or 0):
        raise HTTPException(status_code=403, detail="当前账号不是作者账号")
    return user


def require_admin(request: Request, authorization: str = Header(default="")) -> dict[str, Any]:
    """要求已登录且为管理员身份。"""
    user = require_login(request, authorization)
    if not int(user.get("is_admin") or 0):
        raise HTTPException(status_code=403, detail="管理员权限不足")
    return user


def verify_publish_token(
    x_publish_token: str = Header(default=""),
    resolved_token: str | None = None,
) -> bool:
    """验证桌面端发布令牌。"""
    if not resolved_token:
        raise HTTPException(status_code=503, detail="发布令牌未配置，已拒绝写入")
    if not secrets.compare_digest(x_publish_token, resolved_token):
        raise HTTPException(status_code=401, detail="发布令牌无效")
    return True


_logger = logging.getLogger(__name__)


def try_cloud_sync(request: Request, book_id: int, user: dict | None) -> None:
    """已登录时执行云端同步：静默收集 + 已上传书籍编辑同步。"""
    if not user:
        return
    try:
        sync_service = get_sync_service(request)
        user_id = int(user["id"])
        sync_service.silent_collect(book_id, user_id)
        sync_service.sync_book_edit(book_id, user_id)
    except Exception as e:
        _logger.warning("云端同步失败: book_id=%s, user_id=%s, error=%s", book_id, user.get("id"), e)


def check_book_owner(db: WorkspaceDB, book_id: int, user: dict) -> None:
    """校验书籍归属，不属于当前用户则抛 403。"""
    if not db.check_book_ownership(book_id, int(user["id"])):
        raise HTTPException(status_code=403, detail="无权操作该书籍")

"""
灵枢统一 FastAPI 应用入口。

启动方式:
    uvicorn lingshu_api.main:app --host 0.0.0.0 --port 8787
"""
from __future__ import annotations

import asyncio
import logging
import os
import sys
from contextlib import asynccontextmanager, suppress
from pathlib import Path

from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from slowapi import Limiter
from slowapi.errors import RateLimitExceeded
from slowapi.util import get_remote_address
from starlette.responses import JSONResponse

# 确保项目根目录在 sys.path 中（开发环境）
_project_root = Path(__file__).resolve().parents[1]
if str(_project_root) not in sys.path:
    sys.path.insert(0, str(_project_root))


DEFAULT_DB_PATH = _project_root / "data" / "bookstore_demo.db"


def create_app(db_path: Path | None = None) -> FastAPI:
    """创建并配置 FastAPI 应用。"""
    from lingshu_api.dependencies import set_cloud_store, set_store, set_sync_service
    from lingshu_api.routers import admin, admin_auth, auth, bookshelf, bookstore, workspace
    from lingshu_api.services.book_sync import BookSyncService
    from lingshu_api.services.db_bridge import WorkspaceDBProvider
    from novel_app.bookstore import BookstoreStore, CloudBookStore, _admin_usernames_from_env

    resolved_db_path = db_path or Path(os.environ.get("BOOKSTORE_DB_PATH", "") or DEFAULT_DB_PATH)
    store = BookstoreStore(resolved_db_path)
    store.sync_admin_users(_admin_usernames_from_env())

    cloud_store = CloudBookStore(resolved_db_path)

    class _NonClosingWorkspaceDB:
        def __init__(self, target):
            self._target = target

        def __getattr__(self, name):
            return getattr(self._target, name)

        def close(self):
            return None

    workspace_provider = WorkspaceDBProvider()

    sync_service = BookSyncService(
        cloud_store=cloud_store,
        workspace_db_factory=lambda: _NonClosingWorkspaceDB(workspace_provider.get()),
    )

    # lifespan 替代已废弃的 on_event
    @asynccontextmanager
    async def lifespan(application: FastAPI):
        # startup: 启动 WebSocket 心跳 + 清理过期管理员 Token
        from lingshu_api.ws.connection import get_ws_manager
        ping_task = asyncio.create_task(get_ws_manager().ping_loop())
        try:
            from lingshu_api.routers.admin_auth import _get_db_path, cleanup_on_startup
            cleanup_on_startup(_get_db_path(store))
        except Exception:
            pass
        yield
        # shutdown: 取消心跳任务
        ping_task.cancel()
        with suppress(asyncio.CancelledError):
            await ping_task
        workspace_provider.close()

    _disable_docs = os.environ.get("DISABLE_API_DOCS", "true").lower() in ("true", "1", "yes")
    app = FastAPI(
        title="灵枢 API",
        lifespan=lifespan,
        docs_url=None if _disable_docs else "/docs",
        redoc_url=None if _disable_docs else "/redoc",
        openapi_url=None if _disable_docs else "/openapi.json",
    )

    # 将 store / cloud_store / sync_service 写入 app.state
    set_store(app, store)
    set_cloud_store(app, cloud_store)
    set_sync_service(app, sync_service)
    app.state.workspace_db_provider = workspace_provider

    # 限流
    limiter = Limiter(key_func=get_remote_address)
    app.state.limiter = limiter

    @app.exception_handler(RateLimitExceeded)
    async def _rate_limit_handler(request, exc):
        return JSONResponse(status_code=429, content={"detail": "请求过于频繁，请稍后再试"})

    from fastapi.exceptions import RequestValidationError
    from starlette.exceptions import HTTPException

    @app.exception_handler(HTTPException)
    async def _http_exception_handler(request, exc):
        return JSONResponse(
            status_code=exc.status_code,
            content={"detail": exc.detail},
        )

    @app.exception_handler(RequestValidationError)
    async def _validation_exception_handler(request, exc):
        errors = [
            {
                "field": ".".join(str(loc) for loc in err.get("loc", [])),
                "message": err.get("msg", ""),
                "type": err.get("type", ""),
            }
            for err in exc.errors()
        ]
        return JSONResponse(
            status_code=422,
            content={"detail": "请求参数验证失败", "errors": errors},
        )

    @app.exception_handler(Exception)
    async def _generic_exception_handler(request, exc):
        logger = logging.getLogger("lingshu.api")
        logger.exception("Unhandled exception on %s %s", request.method, request.url.path)
        return JSONResponse(
            status_code=500,
            content={"detail": "服务器内部错误，请稍后重试"},
        )

    # CORS — 生产环境必须通过 CORS_ORIGINS 环境变量配置允许的源
    _cors_env = os.environ.get("CORS_ORIGINS", "").strip()
    if _cors_env:
        allow_origins = [origin.strip() for origin in _cors_env.split(",") if origin.strip()]
    else:
        # 开发环境默认值 — 仅允许已知端口
        allow_origins = [
            "http://127.0.0.1:5173", "http://localhost:5173",
            "http://127.0.0.1:5174", "http://localhost:5174",
            "http://127.0.0.1:5175", "http://localhost:5175",
            "http://127.0.0.1:5176", "http://localhost:5176",
            "https://www.ling-shu.top", "https://ling-shu.top",
        ]

    app.add_middleware(
        CORSMiddleware,
        allow_origins=allow_origins,
        allow_credentials=True,
        allow_methods=["GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"],
        allow_headers=[
            "Authorization",
            "Content-Type",
            "X-Publish-Token",
            "X-Client-KeyId",
            "X-Client-Ts",
            "X-Client-Sig",
        ],
    )

    # 安全响应头中间件
    @app.middleware("http")
    async def add_security_headers(request, call_next):
        response = await call_next(request)
        response.headers["X-Content-Type-Options"] = "nosniff"
        response.headers["X-Frame-Options"] = "DENY"
        response.headers["X-XSS-Protection"] = "1; mode=block"
        response.headers["Referrer-Policy"] = "strict-origin-when-cross-origin"
        response.headers["Permissions-Policy"] = "camera=(), microphone=(), geolocation=()"
        # CSP 仅对页面资源有意义，对 /api/ JSON 响应反而会触发浏览器跨域误判（net::ERR_FAILED）
        path = request.scope.get("path", "")
        if not (path or "").startswith("/api/"):
            response.headers["Content-Security-Policy"] = "default-src 'self'; script-src 'self'; style-src 'self'; img-src 'self' data: https:; connect-src 'self' ws: wss:;"
        return response

    @app.middleware("http")
    async def _admin_hmac_guard(request, call_next):
        path = request.url.path
        # 同时保护 /api/admin/ 和 /api/v1/admin/ 路径
        is_admin_path = (
            path.startswith("/api/admin/") or path.startswith("/api/v1/admin/")
        )
        is_login = path in ("/api/admin/auth/login", "/api/v1/admin/auth/login")
        if is_admin_path and not is_login:
            from fastapi import HTTPException

            from lingshu_api.routers.admin_auth import _admin_keys_configured, verify_admin_hmac_headers

            if not _admin_keys_configured(resolved_db_path):
                return JSONResponse(
                    status_code=503,
                    content={"detail": "管理后台未配置安全密钥，请联系管理员配置 HMAC 密钥后重试"},
                )

            try:
                verify_admin_hmac_headers(resolved_db_path, request)
            except HTTPException as exc:
                return JSONResponse(status_code=exc.status_code, content={"detail": exc.detail})
        return await call_next(request)

    @app.middleware("http")
    async def _api_v1_alias(request, call_next):
        path = str(request.scope.get("path") or "")
        if path.startswith("/api/v1/"):
            request.scope["lingshu.original_path"] = path
            aliased = "/api/" + path.removeprefix("/api/v1/")
            request.scope["path"] = aliased
            request.scope["raw_path"] = aliased.encode("ascii")
        return await call_next(request)

    # API 版本说明:
    # 当前所有 /api/ 路由即为 v1 版本
    # 未来 v2 路由将挂载在 /api/v2/ 前缀下
    # 客户端可通过 /api/v1/ 前缀访问（与 /api/ 等价）
    api_version = "v1"

    # 挂载 REST 路由（/api/ 前缀 — 向后兼容）
    app.include_router(auth.router)
    app.include_router(bookshelf.router)
    app.include_router(bookstore.router)
    app.include_router(admin_auth.router)
    app.include_router(admin.router)
    app.include_router(workspace.router)

    # 挂载阶段2新增路由
    from lingshu_api.routers import (
        analysis_v3,
        characters,
        rag,
        references,
        settings,
        skills,
    )
    from lingshu_api.routers import (
        export as export_router,
    )
    from lingshu_api.routers import (
        world as world_router,
    )
    app.include_router(analysis_v3.router)
    app.include_router(characters.router)
    app.include_router(world_router.router)
    app.include_router(skills.router)
    app.include_router(references.router)
    app.include_router(rag.router)
    app.include_router(export_router.router)
    app.include_router(settings.router)

    # 挂载云端同步路由
    from lingshu_api.routers import cloud_books
    app.include_router(cloud_books.router)

    # 挂载 WebSocket 路由
    from lingshu_api.ws import analysis_v3_ws, chat, generation, review_ws
    app.include_router(generation.router)
    app.include_router(review_ws.router)
    app.include_router(analysis_v3_ws.router)
    app.include_router(chat.router)

    @app.get("/api/health")
    def health():
        return {"ok": True, "version": api_version}

    @app.get("/api/v1/health")
    def health_v1():
        return {"ok": True, "version": api_version}

    return app


# 模块级 app 实例（兼容 uvicorn 直接导入）
app = create_app()

def _require_admin(request: Request, authorization: str, store: BookstoreStore) -> dict[str, Any]:
    token = _bearer_token(authorization)
    if not token:
        raise HTTPException(status_code=401, detail="未登录")
    db_path = _get_db_path(store)
    _ensure_admin_tables(db_path)
    # 双因子验证：HMAC 签名 + Bearer Token
    if _admin_keys_configured(db_path):
        verify_admin_hmac_headers(db_path, request)
    admin = _get_admin_by_token(db_path, token)
    if not admin:
        raise HTTPException(status_code=401, detail="管理员登录已过期或无效")
    return admin

def _cleanup_expired_tokens(db_path: Path) -> int:
    """清理过期的管理员 Token，返回删除数量。"""
    with sqlite3.connect(db_path) as conn:
        cursor = conn.execute(
            f"DELETE FROM {_ADMIN_TOKENS_TABLE} WHERE expires_at <= datetime('now')"

        conn.commit()

def _admin_token_hash(token: str) -> str:
    """Hash an admin token for secure storage."""
    return hashlib.sha256(token.encode("utf-8")).hexdigest()


def _get_db_path(store: BookstoreStore) -> Path:
    return store.db_path


def _get_encryption_key(db_path: Path) -> bytes:
    """获取或生成 HMAC 密钥的 AES-256 加密密钥。
    优先从环境变量 ADMIN_HMAC_ENCRYPTION_KEY 读取（base64 编码的 32 字节密钥），
    否则从数据库同目录的 .admin_encryption_key 文件读取，若不存在则自动生成。"""
    env_key = os.environ.get("ADMIN_HMAC_ENCRYPTION_KEY")
    if env_key:
        return base64.b64decode(env_key)
    p = Path(db_path) if not isinstance(db_path, Path) else db_path
    key_file = p.parent / ".admin_encryption_key"
    if key_file.exists():
        return base64.b64decode(key_file.read_text().strip())
    key = AESGCM.generate_key(bit_length=256)
    key_file.write_text(base64.b64encode(key).decode())
    with contextlib.suppress(OSError):
        os.chmod(str(key_file), 0o600)
    return key


def _encrypt_secret(plaintext: str, db_path: Path) -> str:
    """使用 AES-256-GCM 加密密钥。返回 aes256gcm:<iv>:<ciphertext>:<tag> 格式。"""
    key = _get_encryption_key(db_path)
    aesgcm = AESGCM(key)
    iv = os.urandom(12)
    ciphertext_with_tag = aesgcm.encrypt(iv, plaintext.encode(), None)
    ciphertext = ciphertext_with_tag[:-16]
    tag = ciphertext_with_tag[-16:]
    return f"aes256gcm:{base64.b64encode(iv).decode()}:{base64.b64encode(ciphertext).decode()}:{base64.b64encode(tag).decode()}"


def _decrypt_secret(encrypted: str, db_path: Path) -> str:
    """解密密钥。如果值不以 aes256gcm: 开头，则视为明文（向后兼容）。"""
    if not encrypted.startswith("aes256gcm:"):
        return encrypted
    parts = encrypted.split(":")
    if len(parts) != 4:
        raise ValueError("Invalid encrypted secret format")
    _, iv_b64, ct_b64, tag_b64 = parts
    key = _get_encryption_key(db_path)
    aesgcm = AESGCM(key)
    iv = base64.b64decode(iv_b64)
    ciphertext = base64.b64decode(ct_b64)
    tag = base64.b64decode(tag_b64)
    plaintext = aesgcm.decrypt(iv, ciphertext + tag, None)
    return plaintext.decode()


_ALLOWED_ADMIN_TABLES = frozenset({"admin_users", "admin_auth_tokens", "admin_keys", "admin_audit_logs"})


def _table_columns(connection: sqlite3.Connection, table_name: str) -> set[str]:
    """Get column names for a whitelisted admin table."""
    if table_name not in _ALLOWED_ADMIN_TABLES:
        raise ValueError(f"Table '{table_name}' is not allowed for PRAGMA inspection")
    return {str(row[1]) for row in connection.execute(f"PRAGMA table_info({_sql_identifier(table_name)})").fetchall()}


def _admin_audit_needs_rebuild(connection: sqlite3.Connection) -> bool:
    columns = _table_columns(connection, "admin_audit_logs")
    if not columns:
        return False
    expected = {"id", "admin_user_id", "action", "object_type", "object_id", "detail", "created_at"}
    if not expected.issubset(columns):
        return True
    foreign_keys = connection.execute(f"PRAGMA foreign_key_list({_sql_identifier('admin_audit_logs')})").fetchall()
    return any(str(row[2]) == "users" for row in foreign_keys)


def _column_expr(columns: set[str], *names: str, default: str = "''") -> str:
    for name in names:
        if name in columns:
            return name
    return default


def _ensure_admin_audit_table(connection: sqlite3.Connection) -> None:
    """Normalize admin audit storage so independent admins can write audit rows."""
    connection.execute(
        """
        CREATE TABLE IF NOT EXISTS admin_audit_logs (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            admin_user_id INTEGER NOT NULL,
            action TEXT NOT NULL,
            object_type TEXT NOT NULL,
            object_id TEXT NOT NULL DEFAULT '',
            detail TEXT NOT NULL DEFAULT '',
            created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
        )
        """
    )
    if _admin_audit_needs_rebuild(connection):
        columns = _table_columns(connection, "admin_audit_logs")
        id_expr = _column_expr(columns, "id", default="NULL")
        admin_expr = _column_expr(columns, "admin_user_id", default="0")
        action_expr = _column_expr(columns, "action")
        object_type_expr = _column_expr(columns, "object_type", "target_type")
        object_id_expr = _column_expr(columns, "object_id", "target_id")
        detail_expr = _column_expr(columns, "detail", "detail_json", default="'{}'")
        created_expr = _column_expr(columns, "created_at", default="CURRENT_TIMESTAMP")
        connection.execute("PRAGMA foreign_keys = OFF")
        connection.execute("ALTER TABLE admin_audit_logs RENAME TO admin_audit_logs_old")
        connection.execute(
            """
            CREATE TABLE admin_audit_logs (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                admin_user_id INTEGER NOT NULL,
                action TEXT NOT NULL,
                object_type TEXT NOT NULL,
                object_id TEXT NOT NULL DEFAULT '',
                detail TEXT NOT NULL DEFAULT '',
                created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
            )
            """
        )
        connection.execute(
            f"""
            INSERT INTO admin_audit_logs (id, admin_user_id, action, object_type, object_id, detail, created_at)
            SELECT {id_expr}, {admin_expr}, {action_expr}, {object_type_expr}, {object_id_expr}, {detail_expr}, {created_expr}
            FROM admin_audit_logs_old
            """
        )
        connection.execute("DROP TABLE admin_audit_logs_old")
        connection.execute("PRAGMA foreign_keys = ON")
    connection.execute(
        """
        CREATE INDEX IF NOT EXISTS idx_admin_audit_logs_created
        ON admin_audit_logs(created_at DESC, id DESC)
        """
    )


def _ensure_admin_tables(db_path: Path) -> None:
    with sqlite3.connect(db_path) as connection:
        connection.execute("PRAGMA foreign_keys = ON")
        connection.execute(
            """
            CREATE TABLE IF NOT EXISTS admin_users (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                username TEXT NOT NULL UNIQUE,
                password_hash TEXT NOT NULL,
                display_name TEXT NOT NULL DEFAULT '',
                role TEXT NOT NULL DEFAULT 'operator',
                status TEXT NOT NULL DEFAULT 'active',
                last_login_at TEXT,
                created_at TEXT NOT NULL DEFAULT '',
                updated_at TEXT NOT NULL DEFAULT ''
            )
            """
        )
        connection.execute(
            f"""
            CREATE TABLE IF NOT EXISTS {_ADMIN_TOKENS_TABLE} (
                token TEXT PRIMARY KEY,
                admin_user_id INTEGER NOT NULL,
                expires_at TEXT NOT NULL,
                created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (admin_user_id) REFERENCES admin_users(id) ON DELETE CASCADE
            )
            """
        )
        connection.execute(
            f"""
            CREATE TABLE IF NOT EXISTS {_ADMIN_KEYS_TABLE} (
                key_id TEXT PRIMARY KEY,
                admin_user_id INTEGER NOT NULL,
                secret_encrypted TEXT NOT NULL,
                key_fingerprint TEXT NOT NULL DEFAULT '',
                status TEXT NOT NULL DEFAULT 'active',
                issued_at TEXT NOT NULL DEFAULT '',
                last_used_at TEXT,
                expires_at TEXT,
                FOREIGN KEY (admin_user_id) REFERENCES admin_users(id) ON DELETE CASCADE
            )
            """
        )
        _ensure_admin_audit_table(connection)
        connection.commit()
    # 惰性清理过期 Token
    with contextlib.suppress(Exception):
        _cleanup_expired_tokens(db_path)


def _admin_keys_configured(db_path: Path) -> bool:
    _ensure_admin_tables(db_path)
    with sqlite3.connect(db_path) as connection:
        count = connection.execute(
            f"""
            SELECT COUNT(*)
            FROM {_ADMIN_KEYS_TABLE}
            WHERE status = 'active'
              AND (expires_at IS NULL OR expires_at = '' OR expires_at > datetime('now'))
            """
        ).fetchone()[0]
    return int(count or 0) > 0


def _admin_hmac_message(timestamp: str, method: str, path: str) -> str:
    return f"{timestamp}\n{method.upper()}\n{path}"


def _verify_admin_hmac_signature(
    db_path: Path,
    *,
    key_id: str,
    timestamp: str,
    signature: str,
    method: str,
    path: str,
) -> dict[str, Any] | None:
    _ensure_admin_tables(db_path)
    if not key_id or not timestamp or not signature:
        return None
    try:
        ts = int(timestamp)
    except (TypeError, ValueError):
        return None
    if abs(time.time() * 1000 - ts) > _ADMIN_HMAC_WINDOW_MS:
        return None

    with sqlite3.connect(db_path) as connection:
        connection.row_factory = sqlite3.Row
        row = connection.execute(
            f"""
            SELECT
                k.key_id,
                k.secret_encrypted,
                k.status AS key_status,
                k.expires_at,
                a.id,
                a.username,
                a.display_name,
                a.role,
                a.status AS admin_status,
                a.last_login_at
            FROM {_ADMIN_KEYS_TABLE} AS k
            JOIN admin_users AS a ON a.id = k.admin_user_id
            WHERE k.key_id = ?
            """,
            (key_id,),
        ).fetchone()
        if not row:
            return None
        if str(row["key_status"]) != "active" or str(row["admin_status"]) != "active":
            return None
        expires_at = str(row["expires_at"] or "")
        if expires_at:
            expires = connection.execute("SELECT ? > datetime('now')", (expires_at,)).fetchone()[0]
            if not int(expires or 0):
                return None
        message = _admin_hmac_message(timestamp, method, path)
        secret = _decrypt_secret(str(row["secret_encrypted"]), db_path)
        expected = hmac.new(
            secret.encode("utf-8"),
            message.encode("utf-8"),
            hashlib.sha256,
        ).hexdigest()
        if not hmac.compare_digest(expected, signature):
            return None
        now = datetime.now(UTC).strftime("%Y-%m-%d %H:%M:%S")
        connection.execute(
            f"UPDATE {_ADMIN_KEYS_TABLE} SET last_used_at = ? WHERE key_id = ?",
            (now, key_id),
        )
        connection.commit()
        return {
            "id": row["id"],
            "username": row["username"],
            "display_name": row["display_name"],
            "role": row["role"],
            "status": row["admin_status"],
            "last_login_at": row["last_login_at"],
            "key_id": row["key_id"],
        }


def verify_admin_hmac_headers(db_path: Path, request: Request) -> None:
    if not _admin_keys_configured(db_path):
        raise HTTPException(status_code=503, detail="管理员密钥未配置，请先配置 HMAC 密钥")
    signed_path = str(request.scope.get("lingshu.original_path") or request.url.path)
    verified = _verify_admin_hmac_signature(
        db_path,
        key_id=request.headers.get("X-Client-KeyId", ""),
        timestamp=request.headers.get("X-Client-Ts", ""),
        signature=request.headers.get("X-Client-Sig", ""),
        method=request.method,
        path=signed_path,
    )
    if not verified:
        raise HTTPException(status_code=403, detail="管理密钥签名无效或已过期")


def _get_admin_by_token(db_path: Path, token: str) -> dict[str, Any] | None:
    _ensure_admin_tables(db_path)
    token_hash = _admin_token_hash(token)
    with sqlite3.connect(db_path) as connection:
        connection.row_factory = sqlite3.Row
        row = connection.execute(
            f"""
            SELECT a.id, a.username, a.display_name, a.role, a.status, a.last_login_at
            FROM {_ADMIN_TOKENS_TABLE} AS t
            JOIN admin_users AS a ON a.id = t.admin_user_id
            WHERE t.token = ?
              AND t.expires_at > datetime('now')
              AND a.status = 'active'
            """,
            (token_hash,),
        ).fetchone()
    return dict(row) if row else None


def require_admin_user(
    request: Request,
    authorization: str = Header(default=""),
    store: BookstoreStore = Depends(get_store),
) -> dict[str, Any]:
    token = _bearer_token(authorization)
    if not token:
        raise HTTPException(status_code=401, detail="未登录")
    db_path = _get_db_path(store)
    # 双因子验证：HMAC 签名 + Bearer Token
    verify_admin_hmac_headers(db_path, request)
    admin = _get_admin_by_token(db_path, token)
    if not admin:
        raise HTTPException(status_code=401, detail="管理员登录已过期或无效")
    return admin


@router.post("/login")
@limiter.limit("5/minute")
async def admin_login(request: Request, payload: AdminLoginPayload, store: BookstoreStore = Depends(get_store)) -> dict[str, Any]:
    username = payload.username.strip()
    password = payload.password

    db_path = _get_db_path(store)
    _ensure_admin_tables(db_path)
    if not _admin_keys_configured(db_path):
        raise HTTPException(status_code=503, detail="管理员密钥未配置，请先配置 HMAC 密钥")
    key_info = _verify_admin_hmac_signature(
        db_path,
        key_id=payload.key_id,
        timestamp=payload.timestamp,
        signature=payload.hmac_signature,
        method=request.method,
        path=str(request.scope.get("lingshu.original_path") or request.url.path),
    )
    if not key_info:
        raise HTTPException(status_code=403, detail="密钥验证失败，请检查密钥文件是否有效")
    if str(key_info.get("username") or "") != username:
        raise HTTPException(status_code=403, detail="密钥与管理员账号不匹配")

    with sqlite3.connect(db_path) as connection:
        connection.row_factory = sqlite3.Row
        row = connection.execute(
            """
            SELECT id, username, password_hash, display_name, role, status
            FROM admin_users
            WHERE username = ?
            """,
            (username,),
        ).fetchone()

    if not row or not bcrypt.checkpw(password.encode("utf-8"), str(row["password_hash"]).encode("utf-8")):
        raise HTTPException(status_code=401, detail="管理员用户名或密码错误")
    if str(row["status"]) != "active":
        raise HTTPException(status_code=403, detail="管理员账号已停用")

    token = secrets.token_hex(32)
    now = datetime.now(UTC).strftime("%Y-%m-%d %H:%M:%S")
    expires_at = (datetime.now(UTC) + timedelta(hours=_ADMIN_TOKEN_TTL_HOURS)).strftime("%Y-%m-%d %H:%M:%S")
    with sqlite3.connect(db_path) as connection:
        connection.execute("PRAGMA foreign_keys = ON")
        connection.execute(
            f"""
            INSERT INTO {_ADMIN_TOKENS_TABLE} (token, admin_user_id, expires_at, created_at)
            VALUES (?, ?, ?, ?)
            """,
            (_admin_token_hash(token), int(row["id"]), expires_at, now),
        )
        connection.execute(
            """
            UPDATE admin_users
            SET last_login_at = ?, updated_at = ?
            WHERE id = ?
            """,
            (now, now, int(row["id"])),

    if failure_count > 0 and success_count == 0:
        response_body["ok"] = False
        return JSONResponse(status_code=500, content=response_body)

    return response_body

function writeMainLog(level, message, extra = '') {

//  路径白名单校验：只允许访问 userData 目录
//  使用 realpathSync 解析符号链接，防止通过符号链接绕过白名单
// ------------------------------------------------------------------

function isPathAllowed(filePath) {
  const userDataPath = app.getPath('userData');
  let resolved;
  try {
    resolved = fs.realpathSync(path.resolve(filePath));
  } catch (_err) {
    // 文件不存在时无法解析符号链接，使用 resolve 结果
    resolved = path.resolve(filePath);
  }
  let realUserDataPath;
  try {
    realUserDataPath = fs.realpathSync(userDataPath);
  } catch (_err) {
    realUserDataPath = userDataPath;
  }
  return resolved.startsWith(realUserDataPath + path.sep) || resolved === realUserDataPath;
}

// ------------------------------------------------------------------
//  密钥安全存储 IPC（secure:protect / secure:unprotect）
// ------------------------------------------------------------------

function registerSafeStorageHandlers() {
  ipcMain.handle('secure:protect', (_event, secret) => {
    try {
      if (safeStorage.isEncryptionAvailable()) {
        const encrypted = safeStorage.encryptString(secret);
        return encrypted.toString('base64');
      }
      throw new Error('Electron safeStorage is unavailable; refusing to store secrets without encryption');
    } catch (err) {
      console.error('secure:protect error:', err);
      throw err;
    }
  });

  ipcMain.handle('secure:unprotect', (_event, encryptedBase64) => {
    try {
      if (safeStorage.isEncryptionAvailable()) {
        const buf = Buffer.from(encryptedBase64, 'base64');
        return safeStorage.decryptString(buf);
      }
      throw new Error('Electron safeStorage is unavailable; refusing to read secrets without encryption');
    } catch (err) {
      console.error('secure:unprotect error:', err);
      throw err;
    }
  });
}

//  文件读写 IPC（fs:readFile / fs:writeFile / fs:stat）
//  含大小限制与路径白名单校验

const MAX_FILE_SIZE = 10 * 1024 * 1024; // 10MB

function registerFileIpcHandlers() {
  ipcMain.handle('fs:readFile', async (_event, filePath) => {
    if (typeof filePath !== 'string' || !filePath) {
      throw new Error('Invalid argument: filePath must be a non-empty string');
    }
    if (!isPathAllowed(filePath)) {
      throw new Error('Access denied: path outside allowed directory');
    }
    const stat = await fs.promises.stat(filePath);
    if (stat.size > MAX_FILE_SIZE) {
      throw new Error(`File too large: ${stat.size} bytes (max ${MAX_FILE_SIZE} bytes)`);
    }
    return await fs.promises.readFile(filePath, 'utf-8');
  });

  ipcMain.handle('fs:writeFile', async (_event, filePath, content) => {
    if (typeof filePath !== 'string' || !filePath) {
      throw new Error('Invalid argument: filePath must be a non-empty string');
    }
    if (typeof content !== 'string') {
      throw new Error('Invalid argument: content must be a string');
    }
    if (!isPathAllowed(filePath)) {
      throw new Error('Access denied: path outside allowed directory');
    }
    if (content.length > MAX_FILE_SIZE) {
      throw new Error(`Content too large: ${content.length} bytes (max ${MAX_FILE_SIZE} bytes)`);
    }
    await fs.promises.writeFile(filePath, content, 'utf-8');
    return true;
  });

  ipcMain.handle('fs:stat', async (_event, filePath) => {
    if (typeof filePath !== 'string' || !filePath) {
      throw new Error('Invalid argument: filePath must be a non-empty string');
    }
    if (!isPathAllowed(filePath)) {
      throw new Error('Access denied: path outside allowed directory');
    }
    const stat = await fs.promises.stat(filePath);
    return {
      size: stat.size,
      mtime: stat.mtime.getTime(),
      isFile: stat.isFile(),
      isDirectory: stat.isDirectory(),
    };
  });
}

 *
 * 对应 Qt 模块: app_editor.py → _on_editor_changed / word_timer

import { ref, onUnmounted } from 'vue';

export function useAutoSave(saveFn, options = {}) {
  const { interval = 3000, enabled = true } = options;

  const dirty = ref(false);
  const saving = ref(false);
  const lastSaved = ref(null);
  let timer = null;

  function markDirty() {
    dirty.value = true;
  }

  function markClean() {
    dirty.value = false;
    lastSaved.value = new Date();
  }

  async function save() {
    if (!dirty.value || saving.value) return;
    saving.value = true;
    try {
      await saveFn();
      markClean();
    } finally {
      saving.value = false;
    }
  }

  // 防抖自动保存
  function scheduleAutoSave() {
    if (!enabled || !dirty.value) return;
    clearTimeout(timer);
    timer = setTimeout(() => save(), interval);
  }

  // 关闭时停止
  onUnmounted(() => {
    clearTimeout(timer);
  });

  return { dirty, saving, lastSaved, markDirty, markClean, save, scheduleAutoSave };
}

const listeners = new Map()

export function useEventBus() {
  function on(event, callback) {
    if (!listeners.has(event)) {
      listeners.set(event, new Set())
    }
    listeners.get(event).add(callback)
  }

  function off(event, callback) {
    if (listeners.has(event)) {
      listeners.get(event).delete(callback)
    }
  }

  function emit(event, payload) {
    if (listeners.has(event)) {
      listeners.get(event).forEach(cb => cb(payload))
    }
  }

  return { on, off, emit }
}

import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import api, { getAuthToken, setAuthToken } from '../utils/api';
import { useCloudStore } from './cloud';
import { useEventBus } from '../composables/useEventBus';

export const useAuthStore = defineStore('auth', () => {
  const token = ref(getAuthToken());
  const user = ref(null);
  const initialized = ref(false);
  const loading = ref(false);
  const error = ref('');

  const isLoggedIn = computed(() => !!token.value);
  const isAuthor = computed(() => user.value?.is_author === 1);

  async function login(username, password) {
    return runAuthRequest(async () => {
      const { data } = await api.post('/auth/login', { username, password });
      applySession(data);
      triggerCloudSync();
      return data;
    });
  }

  async function register(username, password, isAuthor = 1) {
    return runAuthRequest(async () => {
      const { data } = await api.post('/auth/register', { username, password, is_author: isAuthor });
      applySession(data);
      triggerCloudSync();
      return data;
    });
  }

  async function fetchMe() {
    if (!token.value) {
      initialized.value = true;
      return null;
    }
    loading.value = true;
    error.value = '';
    try {
      const { data } = await api.get('/auth/me');
      user.value = data;
      initialized.value = true;
      return data;
    } catch (e) {
      error.value = authErrorMessage(e);
      logout();
      return null;
    } finally {
      loading.value = false;
    }
  }

  async function init() {
    if (initialized.value) return user.value;
    if (!token.value) {
      initialized.value = true;
      return null;
    }
    return fetchMe();
  }

  function logout() {
    token.value = '';
    user.value = null;
    error.value = '';
    initialized.value = true;
    setAuthToken('');
  }

  function applySession(data) {
    const accessToken = data?.access_token || '';
    token.value = accessToken;
    user.value = data?.user || null;
    initialized.value = true;
    error.value = '';
    setAuthToken(accessToken);
  }

  async function runAuthRequest(action) {
    loading.value = true;
    error.value = '';
    try {
      return await action();
    } catch (e) {
      error.value = authErrorMessage(e);
      throw e;
    } finally {
      loading.value = false;
    }
  }

  function authErrorMessage(e) {
    const detail = e?.response?.data?.detail;
    if (typeof detail === 'string') return detail;
    if (detail && typeof detail === 'object') return detail.message || '认证失败';
    return e?.message || '认证失败';
  }

  function triggerCloudSync() {
    try {
      const cloudStore = useCloudStore();
      cloudStore.syncUploadedBooks().catch(() => {});
    } catch {
      // cloud store 未初始化时忽略
    }
  }

  function requireAuth() {
    if (isLoggedIn.value) return true;
    const { emit } = useEventBus();
    emit('lingshu:open-login');
    return false;
  }

  function requireAiAccess() {
    return requireAuth();
  }

  return {
    token,
    user,
    initialized,
    loading,
    error,
    isLoggedIn,
    isAuthor,
    init,
    login,
    register,
    fetchMe,
    logout,
    requireAuth,
    requireAiAccess,
  };
});

/**
 * 人物 Store — 管理人物卡列表状态，支持写回事件自动刷新
 */
import { defineStore } from 'pinia';
import { ref } from 'vue';
import api from '../utils/api';
import { useEventBus } from '../composables/useEventBus';

export const useCharacterStore = defineStore('character', () => {
  const characters = ref([]);
  const characterSchemas = ref({ protagonist: [], supporting: [] });
  const loading = ref(false);
  const lastBookId = ref(null);
  let _busListenersBound = false;

  async function fetchCharacters(bookId) {
    if (!bookId) {
      characters.value = [];
      characterSchemas.value = { protagonist: [], supporting: [] };
      return;
    }
    lastBookId.value = bookId;
    loading.value = true;
    try {
      const { data } = await api.get(`/workspace/books/${bookId}/characters`);
      characters.value = data.items || [];
      characterSchemas.value = data.schemas || characterSchemas.value;
    } catch {
      characters.value = [];
    } finally {
      loading.value = false;
    }
  }

  function refresh() {
    if (lastBookId.value) fetchCharacters(lastBookId.value);
  }

  function _bindBusListeners() {
    if (_busListenersBound) return;
    const { on } = useEventBus();
    on('lingshu:characters-updated', (payload) => {
      if (!payload || !lastBookId.value) return;
      if (payload.bookId && Number(payload.bookId) !== Number(lastBookId.value)) return;
      refresh();
    });
    on('lingshu:writeback-completed', (payload) => {
      if (!payload || !lastBookId.value) return;
      if (payload.bookId && Number(payload.bookId) !== Number(lastBookId.value)) return;
      // 仅当写回目标是 character 或批量写回时刷新
      if (payload.writebackTarget && payload.writebackTarget !== 'character') return;
      refresh();
    });
    _busListenersBound = true;
  }

  // 首次调用时绑定事件监听
  _bindBusListeners();

  return { characters, characterSchemas, loading, fetchCharacters, refresh };
});

/**
 * 云端同步 Store — 管理书籍上传/同步状态
 */
import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import api from '../utils/api';

export const useCloudStore = defineStore('cloud', () => {
  // 同步状态: 'idle' | 'uploading' | 'syncing' | 'synced' | 'error'
  const syncStatus = ref('idle');
  const syncError = ref('');
  const lastSyncedAt = ref(null);

  const isSyncing = computed(() => syncStatus.value === 'uploading' || syncStatus.value === 'syncing');

  async function uploadBookToCloud(bookId) {
    // 验证 bookId 为正整数
    if (!bookId || !Number.isInteger(bookId) || bookId <= 0) {
      syncError.value = '无效的书籍ID';
      return;
    }
    syncStatus.value = 'uploading';
    syncError.value = '';
    try {
      const { data } = await api.post(`/cloud/books/${bookId}/upload`);
      syncStatus.value = 'synced';
      lastSyncedAt.value = new Date().toISOString();
      return data;
    } catch (e) {
      syncStatus.value = 'error';
      syncError.value = e?.response?.data?.detail || e?.message || '上传失败';
      throw e;
    }
  }

  async function syncUploadedBooks() {
    syncStatus.value = 'syncing';
    syncError.value = '';
    try {
      const { data } = await api.post('/cloud/sync');
      syncStatus.value = 'synced';
      lastSyncedAt.value = new Date().toISOString();
      return data;
    } catch (e) {
      syncStatus.value = 'error';
      syncError.value = e?.response?.data?.detail || e?.message || '同步失败';
      throw e;
    }
  }

  async function downloadCloudBook(cloudBookId) {
    if (!cloudBookId || !Number.isInteger(cloudBookId) || cloudBookId <= 0) {
      syncError.value = '无效的云端书籍ID';
      return;
    }
    syncStatus.value = 'syncing';
    syncError.value = '';
    try {
      const { data } = await api.post(`/cloud/books/${cloudBookId}/download`);
      syncStatus.value = 'synced';
      lastSyncedAt.value = new Date().toISOString();
      return data;
    } catch (e) {
      syncStatus.value = 'error';
      syncError.value = e?.response?.data?.detail || e?.message || '下载失败';
      throw e;
    }
  }

  async function fetchSyncStatus() {
    try {
      const { data } = await api.get('/cloud/sync/status');
      return data;
    } catch {
      return null;
    }
  }

  function resetStatus() {
    syncStatus.value = 'idle';
    syncError.value = '';
  }

  return {
    syncStatus,
    syncError,
    lastSyncedAt,
    isSyncing,
    uploadBookToCloud,
    downloadCloudBook,
    syncUploadedBooks,
    fetchSyncStatus,
    resetStatus,
  };
});

/**
 * 任务 Store — AI 任务状态管理，占位
 */
import { defineStore } from 'pinia';
import { ref } from 'vue';

export const useTaskStore = defineStore('task', () => {
  const tasks = ref([]);

  function addTask(task) {
    tasks.value.unshift({ id: Date.now(), status: 'pending', ...task });
  }

  function updateTask(taskId, updates) {
    const idx = tasks.value.findIndex(t => t.id === taskId);
    if (idx >= 0) Object.assign(tasks.value[idx], updates);
  }

  function clearCompleted() {
    tasks.value = tasks.value.filter(t => t.status !== 'completed');
  }

  return { tasks, addTask, updateTask, clearCompleted };
});

/**
 * 主题 Store — 4 套预设主题切换
 */
import { defineStore } from 'pinia';
import { ref } from 'vue';

const THEMES = {
  white: { label: '白昼', className: '' },
  'light-blue': { label: '浅蓝', className: 'theme-light-blue' },
  night: { label: '暗夜', className: 'theme-night' },
  sepia: { label: '羊皮纸', className: 'theme-sepia' },
};

export const useThemeStore = defineStore('theme', () => {
  const saved = localStorage.getItem('lingshu_theme') || 'white';
  const current = ref(saved in THEMES ? saved : 'white');

  const themes = THEMES;
  const currentLabel = ref(THEMES[current.value].label);

  function apply(newTheme) {
    if (!(newTheme in THEMES)) return;
    // 移除旧主题 class
    Object.values(THEMES).forEach(t => {
      if (t.className) document.documentElement.classList.remove(t.className);
    });
    // 应用新主题
    current.value = newTheme;
    currentLabel.value = THEMES[newTheme].label;
    const cls = THEMES[newTheme].className;
    if (cls) document.documentElement.classList.add(cls);
    localStorage.setItem('lingshu_theme', newTheme);
  }

  function toggle() {
    const keys = Object.keys(THEMES);
    const idx = keys.indexOf(current.value);
    apply(keys[(idx + 1) % keys.length]);
  }

  // 初始化时应用保存的主题
  apply(current.value);

  return { current, currentLabel, themes, apply, toggle };
});

/**
 * WebSocket Store — 连接管理（含自动重连）
 */
import { defineStore } from 'pinia';
import { ref } from 'vue';
import { buildWsUrl, sendWsAuth } from '../utils/api';

const RECONNECT_DELAY_MS = 3000;
const MAX_RECONNECT_ATTEMPTS = 5;

export const useWsStore = defineStore('ws', () => {
  const connections = ref({});
  const status = ref({});

  function connect(url) {
    if (connections.value[url]) return connections.value[url];

    const ws = new WebSocket(buildWsUrl(url));
    let reconnectAttempts = 0;
    let intentionalClose = false;

    ws.onopen = () => {
      status.value[url] = 'connected';
      reconnectAttempts = 0;
      // 重连后重新发送认证消息
      sendWsAuth(ws);
    };
    ws.onclose = () => {
      status.value[url] = 'disconnected';
      delete connections.value[url];
      // 主动断开时不重连
      if (intentionalClose) return;
      // 自动重连（指数退避）
      if (reconnectAttempts < MAX_RECONNECT_ATTEMPTS) {
        reconnectAttempts++;
        const delay = RECONNECT_DELAY_MS * Math.pow(2, reconnectAttempts - 1);
        setTimeout(() => {
          if (!connections.value[url]) {
            connect(url);
          }
        }, delay);
      }
    };
    ws.onerror = () => { status.value[url] = 'error'; };

    connections.value[url] = ws;
    status.value[url] = 'connecting';
    // 保存 intentionalClose 标志以便 disconnect 使用
    ws._setIntentionalClose = () => { intentionalClose = true; };
    return ws;
  }

  function disconnect(url) {
    const ws = connections.value[url];
    if (ws) {
      if (ws._setIntentionalClose) ws._setIntentionalClose();
      ws.close();
      delete connections.value[url];
    }
  }

  function send(url, message) {
    const ws = connections.value[url];
    if (ws && ws.readyState === WebSocket.OPEN) {
      ws.send(JSON.stringify(message));
    }
  }

  function disconnectAll() {
    Object.keys(connections.value).forEach(disconnect);
  }

  return { connections, status, connect, disconnect, send, disconnectAll };
});

import axios from 'axios';
import { useEventBus } from '../composables/useEventBus';

const electronBridge = typeof window !== 'undefined' ? window.lingshu : null;

export const API_BASE_URL = import.meta.env.VITE_API_BASE_URL
  || electronBridge?.apiBaseUrl
  || 'http://127.0.0.1:8787/api';

export const WS_BASE_URL = import.meta.env.VITE_WS_BASE_URL
  || electronBridge?.wsBaseUrl
  || 'ws://127.0.0.1:8787/ws';

function _sameOriginWsBase() {
  if (typeof location === 'undefined' || !location.host) return null;
  const protocol = location.protocol === 'https:' ? 'wss:' : 'ws:';
  return `${protocol}//${location.host}/ws`;
}

function _normalizeWsUrl(rawUrl) {
  const url = new URL(rawUrl);
  if (!['ws:', 'wss:'].includes(url.protocol)) {
    throw new Error('WebSocket URL must use ws:// or wss://');
  }
  if (url.username || url.password) {
    throw new Error('WebSocket URL must not include credentials');
  }
  if (!url.pathname.startsWith('/ws')) {
    throw new Error('WebSocket URL path must start with /ws');
  }
  return url;
}

function _allowedWsOrigins() {
  const origins = new Set();
  try {
    origins.add(_normalizeWsUrl(WS_BASE_URL).origin);
  } catch {
    // Invalid configured base URL will be reported when buildWsUrl is used.
  }
  const sameOriginBase = _sameOriginWsBase();
  if (sameOriginBase) {
    origins.add(new URL(sameOriginBase).origin);
  }
  return origins;
}

function _assertAllowedWsUrl(rawUrl) {
  const url = _normalizeWsUrl(rawUrl);
  if (!_allowedWsOrigins().has(url.origin)) {
    throw new Error(`WebSocket URL origin is not allowed: ${url.origin}`);
  }
  return url.toString();
}

export const api = axios.create({
  baseURL: API_BASE_URL,
  timeout: 0,
});

// ============================================================
//  Token 安全存储 — Electron 环境使用 safeStorage 加密
//  策略：优先内存 Token，Electron 环境仅使用 safeStorage 加密存储；
//        非 Electron 环境只保存在内存中，刷新后重新登录。
// ============================================================

let _memoryToken = '';

// Token 初始化就绪 Promise：防止异步解密未完成时请求发出 401
let _tokenReadyResolve;
const tokenReady = new Promise((resolve) => { _tokenReadyResolve = resolve; });

function _initTokenFromSafeStorage() {
  if (!electronBridge?.unprotectSecret) {
    localStorage.removeItem('lingshu_token');
    _tokenReadyResolve();
    return;
  }
  const encrypted = localStorage.getItem('lingshu_token_encrypted');
  if (encrypted) {
    electronBridge.unprotectSecret(encrypted).then((decrypted) => {
      if (decrypted) {
        _memoryToken = decrypted;
      }
      _tokenReadyResolve();
    }).catch(() => {
      // safeStorage 解密失败，不回退到明文 Token，强制重新登录
      localStorage.removeItem('lingshu_token_encrypted');
      localStorage.removeItem('lingshu_token');
      _tokenReadyResolve();
    });
  } else {
    localStorage.removeItem('lingshu_token');
    _tokenReadyResolve();
  }
}

function _migrateToSafeStorage(token) {
  localStorage.removeItem('lingshu_token');
  if (!electronBridge?.protectSecret || !token) return;
  electronBridge.protectSecret(token).then((encrypted) => {
    localStorage.setItem('lingshu_token_encrypted', encrypted);
    localStorage.removeItem('lingshu_token');
  }).catch(() => {
    // safeStorage 加密失败时只保留内存 token，不回退到明文 localStorage。
    localStorage.removeItem('lingshu_token_encrypted');
    localStorage.removeItem('lingshu_token');
  });
}

function _clearTokenStorage() {
  _memoryToken = '';
  localStorage.removeItem('lingshu_token');
  localStorage.removeItem('lingshu_token_encrypted');
}

// 初始化：从安全存储加载 token
_initTokenFromSafeStorage();

export function getAuthToken() {
  return _memoryToken || '';
}

export function setAuthToken(token) {
  if (token) {
    _memoryToken = token;
    if (electronBridge?.protectSecret) {
      // Electron 环境：仅使用 safeStorage 加密存储，不写明文 localStorage
      _migrateToSafeStorage(token);
    } else {
      localStorage.removeItem('lingshu_token');
    }
  } else {
    _clearTokenStorage();
  }
}

export function buildWsUrl(path) {
  const value = String(path || '');
  if (value.startsWith('ws://') || value.startsWith('wss://')) {
    return _assertAllowedWsUrl(value);
  }
  if (/^[a-z][a-z0-9+.-]*:/i.test(value)) {
    throw new Error('WebSocket URL must be a /ws path or ws(s) URL');
  }
  const base = _normalizeWsUrl(WS_BASE_URL || _sameOriginWsBase());
  const basePath = base.pathname.replace(/\/+$/, '');
  const childPath = value.startsWith('/') ? value : `/${value}`;
  const finalPath = childPath === basePath || childPath.startsWith(`${basePath}/`)
    ? childPath
    : `${basePath}${childPath}`;
  return _assertAllowedWsUrl(`${base.protocol}//${base.host}${finalPath}`);
}

export function sendWsAuth(ws) {
  const token = getAuthToken();
  ws.send(JSON.stringify({ type: 'auth', payload: { token } }));
  return token;
}

// 请求拦截器：等待 Token 就绪后自动附加 Authorization
api.interceptors.request.use(async (config) => {
  await tokenReady;
  const token = getAuthToken();
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// 响应拦截器：统一错误处理
api.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response?.status === 401) {
      _clearTokenStorage();
      const url = String(error.config?.url || '');
      const isAuthRequest = url.includes('/auth/login') || url.includes('/auth/register');
      if (!isAuthRequest && typeof window !== 'undefined') {
        const { emit } = useEventBus();
        emit('lingshu:auth-expired');
        emit('lingshu:open-login');
      }
    }
    return Promise.reject(error);
  },
);

export default api;
