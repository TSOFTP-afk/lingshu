   12 │ @dataclass(frozen=True)
   13 │ class TextSpan:
   14 │     start: int
   15 │     end: int
   16 │     text: str
   17 │     reason: str

   20 │ class SemanticChunker:
   21 │     def __init__(
   22 │         self,
   23 │         *,
   24 │         target_chars: int = 6000,
   25 │         max_chars: int = 8000,
   26 │         min_sentence_cut_chars: int = 1200,
   27 │         overlap_sentences: int = 1,
   28 │     ) -> None:
   29 │         self.target_chars = max(1, int(target_chars))
   30 │         self.max_chars = max(self.target_chars, int(max_chars))
   31 │         self.min_sentence_cut_chars = max(200, int(min_sentence_cut_chars))
   32 │         self.overlap_sentences = max(0, int(overlap_sentences))
   33 │ 
   34 │     def chunk_chapters(self, book_id: int, chapters: list[dict[str, Any]], *, pipeline_version: str) -> list[Semantic...
   35 │         blocks: list[SemanticBlock] = []
   36 │         for chapter in chapters:
   37 │             chapter_id = self._int_or_none(chapter.get("id"))
   38 │             title = str(chapter.get("title") or "")
   39 │             content = str(chapter.get("content") or "")
   40 │             if not content.strip():
   41 │                 continue
   42 │             chapter_content_hash = stable_hash(content)
   43 │             local_index = 1
   44 │             for span in self._chapter_spans(content):
   45 │                 block = SemanticBlock.from_text_span(
   46 │                     book_id=book_id,
   47 │                     chapter_id=chapter_id,
   48 │                     chapter_title=title,
   49 │                     local_index=local_index,
   50 │                     chunk_index=len(blocks) + 1,
   51 │                     char_start=span.start,
   52 │                     char_end=span.end,
   53 │                     text=span.text,
   54 │                     pipeline_version=pipeline_version,
   55 │                     split_reason=span.reason,
   56 │                     chapter_content_hash=chapter_content_hash,
   57 │                 )
   58 │                 blocks.append(block)
   59 │                 local_index += 1
   60 │ 
   61 │         # 句子级重叠窗口：为每个 block 添加前一个 block 末尾的 overlap_sentences 个句子
   62 │         if self.overlap_sentences > 0 and len(blocks) > 1:
   63 │             blocks = self._apply_overlap_window(blocks)
   64 │ 
   65 │         return blocks
   66 │ 
   67 │     def _chapter_spans(self, content: str) -> list[TextSpan]:
   68 │         source = str(content or "")
   69 │         scene_spans = self._scene_spans(source)
   70 │         blocks: list[TextSpan] = []
   71 │         for scene in scene_spans:
   72 │             pieces = self._paragraph_spans(source, scene.start, scene.end)
   73 │             if not pieces:
   74 │                 continue
   75 │             current: list[TextSpan] = []
   76 │ 
   77 │             def flush(reason: str) -> None:
   78 │                 nonlocal current
   79 │                 if not current:
   80 │                     return
   81 │                 start = current[0].start
   82 │                 end = current[-1].end
   83 │                 blocks.extend(self._finalize_span(source, start, end, reason))
   84 │                 current = []
   85 │ 
   86 │             for piece in pieces:
   87 │                 if len(piece.text) > self.max_chars:
   88 │                     flush("paragraph_boundary")
   89 │                     blocks.extend(self._split_long_span(source, piece.start, piece.end))

  166 │     def _apply_overlap_window(self, blocks: list[SemanticBlock]) -> list[SemanticBlock]:
  167 │         """为每个 block 添加前一个 block 末尾的 overlap_sentences 个句子作为重叠上下文。

  200 │     @staticmethod
  201 │     def _extract_trailing_sentences(text: str, count: int) -> str:
  202 │         """从文本末尾提取最后 count 个句子。"""
  203 │         if not text or count <= 0:
  204 │             return ""
  205 │         # 按中文/英文句号、感叹号、问号分句
  206 │         sentences = re.split(r"(?<=[。！？.!?])", text)

  208 │         if not sentences:
  209 │             return ""
  210 │         trailing = sentences[-count:]

  205 │     def _phase3_cross_chapter_merge(

  207 │         buckets: dict[tuple[str, str], list[dict[str, Any]]],
  208 │         book_id: int,
  209 │     ) -> dict[tuple[str, str], list[dict[str, Any]]]:
  210 │         """合并跨章的同实体候选。
  211 │

  214 │         2. 名称相似度超过阈值（编辑距离），或一方是另一方的别名/子串

  217 │         keys = list(buckets.keys())
  218 │         merged: set[tuple[str, str]] = set()
  219 │         merges: list[tuple[tuple[str, str], tuple[str, str]]] = []
  220 │ 
  221 │         for i, key_a in enumerate(keys):
  222 │             if key_a in merged:
  223 │                 continue
  224 │             for j in range(i + 1, len(keys)):

  226 │                 if key_b in merged:
  227 │                     continue
  228 │                 if key_a[0] != key_b[0]:
  229 │                     continue

  231 │                     merges.append((key_a, key_b))
  232 │                     merged.add(key_b)

  234 │         # 执行合并
  235 │         for key_a, key_b in merges:
  236 │             buckets[key_a].extend(buckets.pop(key_b, []))
  237 │ 
  238 │         return buckets

  240 │     def _should_merge_buckets(
  241 │         self,
  242 │         bucket_a: list[dict[str, Any]],
  243 │         bucket_b: list[dict[str, Any]],
  244 │     ) -> bool:
  245 │         """判断两个 bucket 是否应该合并。"""
  246 │         names_a = {str(item.get("quote_text") or "").strip() for item in bucket_a}
  247 │         names_b = {str(item.get("quote_text") or "").strip() for item in bucket_b}
  248 │ 
  249 │         # 条件 1: 别名显式 — 一方的 quote 是另一方的子串或别名
  250 │         for name_a in names_a:
  251 │             for name_b in names_b:
  252 │                 if not name_a or not name_b:
  253 │                     continue
  254 │                 if name_a == name_b:
  255 │                     return True
  256 │                 short, long = (name_a, name_b) if len(name_a) < len(name_b) else (name_b, name_a)
  257 │                 if len(short) >= 2 and long.startswith(short):
  258 │                     return True
  259 │ 
  260 │         # 条件 2: 名称相似度
  261 │         for name_a in names_a:
  262 │             for name_b in names_b:
  263 │                 if not name_a or not name_b:
  264 │                     continue
  265 │                 if _name_similarity(name_a, name_b) >= (1.0 - _NAME_SIMILARITY_THRESHOLD):
  266 │                     return True
  267 │ 
  268 │         # 条件 3: 向量相似度辅助 — 如果两个 bucket 的上下文向量高度相似，也合并
  269 │         if self._vector_store is not None:
  270 │             try:
  271 │                 vector_score = self._vector_similarity_score(bucket_a, bucket_b)
  272 │                 if vector_score >= 0.75:
  273 │                     return True
  274 │             except Exception:
  275 │                 pass
  276 │ 
  277 │         return False

  279 │     def _vector_similarity_score(
  280 │         self,
  281 │         bucket_a: list[dict[str, Any]],
  282 │         bucket_b: list[dict[str, Any]],
  283 │     ) -> float:
  284 │         """利用向量存储计算两个 bucket 的上下文相似度。"""
  285 │         if not self._vector_store:
  286 │             return 0.0
  287 │         # 取每个 bucket 的第一个 mention 的 quote_text 作为查询
  288 │         quote_a = str(bucket_a[0].get("quote_text") or "").strip() if bucket_a else ""
  289 │         quote_b = str(bucket_b[0].get("quote_text") or "").strip() if bucket_b else ""
  290 │         if not quote_a or not quote_b:
  291 │             return 0.0
  292 │         book_id = int(bucket_a[0].get("book_id") or 0)
  293 │         if not book_id:
  294 │             return 0.0
  295 │         # 用 quote_a 搜索 entity_summaries，看是否命中 bucket_b 的实体
  296 │         results_a = self._vector_store.search_similar_entities(book_id, quote_a, top_k=3, score_threshold=0.5)
  297 │         results_b = self._vector_store.search_similar_entities(book_id, quote_b, top_k=3, score_threshold=0.5)
  298 │         # 如果两者的搜索结果有重叠（命中同一实体），说明上下文相似
  299 │         uids_a = {r.get("payload", {}).get("entity_uid") for r in results_a}
  300 │         uids_b = {r.get("payload", {}).get("entity_uid") for r in results_b}
  301 │         overlap = uids_a & uids_b
  302 │         if overlap:
  303 │             # 取重叠实体的最高分数
  304 │             max_score = 0.0
  305 │             for r in results_a + results_b:
  306 │                 if r.get("payload", {}).get("entity_uid") in overlap:
  307 │                     max_score = max(max_score, r.get("score", 0.0))
  308 │             return min(1.0, max_score)
  309 │         return 0.0

  332 │ # 工具函数

  335 │ def _name_similarity(a: str, b: str) -> float:
  336 │     """计算两个名称的相似度（0~1），基于编辑距离。"""
  337 │     if not a or not b:
  338 │         return 0.0
  339 │     if a == b:
  340 │         return 1.0
  341 │     dist = _edit_distance(a, b)
  342 │     max_len = max(len(a), len(b))
  343 │     return 1.0 - (dist / max_len)
  344 │ 
  345 │ 
  346 │ def _edit_distance(a: str, b: str) -> int:
  347 │     """Levenshtein 编辑距离。"""
  348 │     n, m = len(a), len(b)
  349 │     if n == 0:
  350 │         return m
  351 │     if m == 0:
  352 │         return n
  353 │     prev = list(range(m + 1))
  354 │     curr = [0] * (m + 1)
  355 │     for i in range(1, n + 1):
  356 │         curr[0] = i
  357 │         for j in range(1, m + 1):
  358 │             cost = 0 if a[i - 1] == b[j - 1] else 1
  359 │             curr[j] = min(prev[j] + 1, curr[j - 1] + 1, prev[j - 1] + cost)
  360 │         prev, curr = curr, prev
  361 │     return prev[m]
  362 │

  364 │ def _int_or_none(value: Any) -> int | None:
  365 │     try:
  366 │         result = int(value)
  367 │     except (TypeError, ValueError):
  368 │         return None
  369 │     return result if result > 0 else None

  277 │ def _verify_admin_hmac_signature(
  278 │     db_path: Path,
  279 │     *,
  280 │     key_id: str,
  281 │     timestamp: str,
  282 │     signature: str,
  283 │     method: str,
  284 │     path: str,
  285 │ ) -> dict[str, Any] | None:
  286 │     _ensure_admin_tables(db_path)
  287 │     if not key_id or not timestamp or not signature:
  288 │         return None
  289 │     try:
  290 │         ts = int(timestamp)
  291 │     except (TypeError, ValueError):
  292 │         return None
  293 │     if abs(time.time() * 1000 - ts) > _ADMIN_HMAC_WINDOW_MS:
  294 │         return None
  295 │ 
  296 │     with sqlite3.connect(db_path) as connection:
  297 │         connection.row_factory = sqlite3.Row
  298 │         row = connection.execute(
  299 │             f"""
  300 │             SELECT
  301 │                 k.key_id,
  302 │                 k.secret_encrypted,
  303 │                 k.status AS key_status,
  304 │                 k.expires_at,
  305 │                 a.id,
  306 │                 a.username,
  307 │                 a.display_name,
  308 │                 a.role,
  309 │                 a.status AS admin_status,
  310 │                 a.last_login_at
  311 │             FROM {_ADMIN_KEYS_TABLE} AS k
  312 │             JOIN admin_users AS a ON a.id = k.admin_user_id
  313 │             WHERE k.key_id = ?
  314 │             """,
  315 │             (key_id,),
  316 │         ).fetchone()

  318 │             return None
  319 │         if str(row["key_status"]) != "active" or str(row["admin_status"]) != "active":
  320 │             return None
  321 │         expires_at = str(row["expires_at"] or "")
  322 │         if expires_at:
  323 │             expires = connection.execute("SELECT ? > datetime('now')", (expires_at,)).fetchone()[0]
  324 │             if not int(expires or 0):
  325 │                 return None
  326 │         message = _admin_hmac_message(timestamp, method, path)
  327 │         secret = _decrypt_secret(str(row["secret_encrypted"]), db_path)
  328 │         expected = hmac.new(
  329 │             secret.encode("utf-8"),
  330 │             message.encode("utf-8"),
  331 │             hashlib.sha256,
  332 │         ).hexdigest()
  333 │         if not hmac.compare_digest(expected, signature):
  334 │             return None
  335 │         now = datetime.now(UTC).strftime("%Y-%m-%d %H:%M:%S")
  336 │         connection.execute(
  337 │             f"UPDATE {_ADMIN_KEYS_TABLE} SET last_used_at = ? WHERE key_id = ?",
  338 │             (now, key_id),
  339 │         )
  340 │         connection.commit()
  341 │         return {
  342 │             "id": row["id"],
  343 │             "username": row["username"],
  344 │             "display_name": row["display_name"],
  345 │             "role": row["role"],
  346 │             "status": row["admin_status"],
  347 │             "last_login_at": row["last_login_at"],
  348 │             "key_id": row["key_id"],
  349 │         }
